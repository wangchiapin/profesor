    // ===================================================================
    // ===== 步驟 2：比比看「練習範圍」（老師管理中心）=====
    // 貼在中心專案 subscription-48d3c 的規則裡：class_plays 那段之後、最後兩個 } 之前。
    // 依賴既有函式：signedIn() / realUser() / isAdmin() / subActive(uid)（跟比比看 class_plays 規則用的同一組）
    //
    // bbk_sets/{uid}   老師自己的範圍清單 { items:[{id,name,lv,d,t,n,sid?}], updatedAt }
    //   - 只有本人讀寫
    //   - 額度：免費 3 組、訂閱中 15 組；到期後超過的仍可改名、修改、刪除（只要數量不增加）
    // shared_bbk/{sid} 給學生連結 ?r=<sid> 讀的範圍 { owner,name,th,lv,d,t,n,updatedAt }
    //   - 任何人都可以讀單筆（學生不用登入）；不能列出全部
    //   - 只有建立的老師能改、刪
    // ===================================================================
    match /bbk_sets/{uid} {
      function okList() {
        let d = request.resource.data;
        let n = d.items.size();
        return d.keys().hasOnly(['items','updatedAt'])
          && d.items is list
          && d.updatedAt == request.time
          && (n <= 3
              || (n <= 15 && subActive(uid))
              || (resource != null && n <= resource.data.items.size()));
      }
      allow get: if realUser() && (request.auth.uid == uid || isAdmin());
      allow create, update: if realUser() && request.auth.uid == uid && okList();
      allow delete: if realUser() && (request.auth.uid == uid || isAdmin());
    }

    match /shared_bbk/{sid} {
      function okSet() {
        let d = request.resource.data;
        return d.keys().hasOnly(['owner','name','th','lv','d','t','n','updatedAt'])
          && d.keys().hasAll(['owner','name','th','lv','d','t','n','updatedAt'])
          && d.owner == request.auth.uid
          && d.name is string && d.name.size() >= 1 && d.name.size() <= 40
          && d.th is string && d.th.size() <= 20
          && d.lv is int && d.lv >= 1 && d.lv <= 3
          && d.d is list && d.d.size() <= 20
          && d.t is list && d.t.size() <= 20
          && d.n is int && d.n >= 3 && d.n <= 30
          && d.updatedAt == request.time;
      }
      allow get: if true;
      allow list: if false;
      allow create: if realUser() && sid.matches('^[a-z0-9]{12}$') && okSet();
      allow update: if realUser() && resource.data.owner == request.auth.uid && okSet();
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }
