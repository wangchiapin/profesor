    // 老師自己的墨龍設定（自訂主題、練習範圍）
    match /dragon_sets/{uid} {
      allow get: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['themes', 'sets', 'updatedAt'])
        && request.resource.data.themes is list && request.resource.data.themes.size() <= 3
        && request.resource.data.sets is list && request.resource.data.sets.size() <= 15
        // 要訂閱才能新增或修改；沒訂閱只能縮減（刪除主題或範圍）
        && (subActive(uid)
            || (resource != null
                && request.resource.data.themes.size() <= resource.data.themes.size()
                && request.resource.data.sets.size() <= resource.data.sets.size()));
      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    // 老師發給學生的「練習範圍」連結（?d=碼）：知道碼的人才讀得到，不能列出全部
    match /shared_dragon/{sid} {
      allow get: if true;
      allow list: if false;
      allow create, update: if request.auth != null
        && sid.matches('^[a-z0-9]{12}$')
        && request.resource.data.owner == request.auth.uid
        && (resource == null || resource.data.owner == request.auth.uid)
        && request.resource.data.keys().hasOnly(['owner', 'name', 'builtin', 'custom', 'updatedAt'])
        && request.resource.data.builtin is list && request.resource.data.builtin.size() <= 10
        && request.resource.data.custom is list && request.resource.data.custom.size() <= 3
        && subActive(request.auth.uid);
      allow delete: if request.auth != null && resource.data.owner == request.auth.uid;
    }
