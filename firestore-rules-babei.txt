// 整份貼進中心專案 subscription-48d3c 的規則即可（class_plays 這次沒動；步驟 3 做錯題明細時再改）。
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ===== 共用函式 =====
    function signedIn() { return request.auth != null; }
    function realUser() {
      return request.auth != null && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email == 'a2224907@gmail.com'
        && request.auth.token.email_verified == true;
    }

    function subActive(uid) {
      return exists(/databases/$(database)/documents/site_plans/$(uid))
        && get(/databases/$(database)/documents/site_plans/$(uid)).data.plan in ['month','year','trial','pro']
        && get(/databases/$(database)/documents/site_plans/$(uid)).data.expiresAt != null
        && get(/databases/$(database)/documents/site_plans/$(uid)).data.expiresAt > request.time;
    }

    // ===== 管理員可讀寫全部 =====
    match /{document=**} {
      allow read, write: if isAdmin();
    }

    // ===== 入口網站 =====
    match /portalStats/{docId} {
      allow read: if true;
      allow create: if statId(docId)
        && request.resource.data.keys().hasOnly(['n'])
        && request.resource.data.n == 1;
      allow update: if statId(docId)
        && request.resource.data.keys().hasOnly(['n'])
        && request.resource.data.n == resource.data.n + 1;
      allow delete: if false;
    }
    match /portalStatus/{docId} { allow read: if true; }
    match /portalPublic/{docId} { allow read: if true; }
    match /portalNotes/{noteId} {
      allow create: if request.resource.data.keys().hasOnly(['name', 'message', 'lang', 'createdAt'])
        && request.resource.data.keys().hasAll(['name', 'message', 'lang', 'createdAt'])
        && request.resource.data.name is string && request.resource.data.name.size() <= 40
        && request.resource.data.message is string && request.resource.data.message.size() <= 500
        && (request.resource.data.name.size() > 0 || request.resource.data.message.size() > 0)
        && request.resource.data.lang in ['zh', 'es']
        && request.resource.data.createdAt == request.time;
    }
    match /portalFeedback/{fbId} {
      allow create: if request.resource.data.keys().hasOnly(['type', 'target', 'app', 'kind', 'message', 'name', 'contact', 'lang', 'createdAt'])
        && request.resource.data.keys().hasAll(['type', 'target', 'app', 'kind', 'message', 'name', 'contact', 'lang', 'createdAt'])
        && request.resource.data.type in ['game', 'font', 'tool']
        && request.resource.data.target is string && request.resource.data.target.size() <= 40
        && request.resource.data.app in ['', 'word', 'ppt', 'other']
        && request.resource.data.kind in ['suggest', 'bug', 'thoughts', 'pinyin', 'install']
        && request.resource.data.message is string
        && request.resource.data.message.size() > 0 && request.resource.data.message.size() <= 2000
        && request.resource.data.name is string && request.resource.data.name.size() <= 40
        && request.resource.data.contact is string && request.resource.data.contact.size() <= 100
        && request.resource.data.lang in ['zh', 'es']
        && request.resource.data.createdAt == request.time;
    }
    match /portalVotes/{voteId} { allow read: if true; }
    match /portalVoteCount/{docId} {
      allow read: if true;
      allow create: if voteCountId(docId)
        && request.resource.data.keys().hasOnly(['n'])
        && request.resource.data.n == 1;
      allow update: if voteCountId(docId)
        && request.resource.data.keys().hasOnly(['n'])
        && request.resource.data.n == resource.data.n + 1;
      allow delete: if false;
    }
    function voteCountId(id) {
      return id.matches('[A-Za-z0-9]{20}__(agree|disagree)');
    }
    function statId(id) {
      return id.matches('(total|d_[0-9]{4}-[0-9]{2}-[0-9]{2})__(views|visitors|(game|download|support|tool|toolview|toolcopy)_[a-z0-9-]{1,40})');
    }

    // ===== 訂閱中心 =====
    match /site_members/{uid} {
      allow get: if signedIn() && request.auth.uid == uid;
      allow create: if realUser() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['email','name','createdAt'])
        && request.resource.data.keys().hasAll(['email','name','createdAt'])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.name is string && request.resource.data.name.size() <= 100
        && request.resource.data.createdAt == request.time;
    }

    match /site_plans/{uid} {
      allow get: if signedIn() && request.auth.uid == uid;
    }

    match /site_requests/{id} {
      allow create: if realUser()
        && request.resource.data.keys().hasOnly(['uid','plan','method','currency','note','ref','paidAt','status','createdAt'])
        && request.resource.data.keys().hasAll(['uid','plan','method','currency','note','ref','status','createdAt'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.ref is string && request.resource.data.ref.size() <= 80
        && (request.resource.data.method != 'paypal' || request.resource.data.ref.size() >= 6)
        && (request.resource.data.method != 'twd' || request.resource.data.ref.matches('^[0-9]{5}$'))
        && (!('paidAt' in request.resource.data) || (request.resource.data.paidAt is string && request.resource.data.paidAt.size() <= 32))
        && request.resource.data.plan in ['month','year']
        && request.resource.data.method in ['paypal','twd','linepay','mxn']
        && request.resource.data.currency in ['TWD','USD','MXN']
        && request.resource.data.note is string && request.resource.data.note.size() <= 300
        && request.resource.data.status == 'pending'
        && request.resource.data.createdAt == request.time;
      allow read: if realUser() && resource.data.uid == request.auth.uid;
    }

    match /site_catalog/{itemId} { allow read: if true; }

    match /site_config/{docId} { allow read: if true; }

    match /site_content/{id} {
      allow read: if realUser() && subActive(request.auth.uid);
    }

    // ===== 色數消消樂：設定組 =====
    function planLimit(uid) {
      return subActive(uid) ? 15 : 3;
    }

    match /teacher_configs/{uid} {
      allow read, delete: if signedIn() && request.auth.uid == uid;
      allow create, update: if realUser() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['items','updatedAt'])
        && request.resource.data.items is list
        && request.resource.data.items.size() <= planLimit(uid)
        && request.resource.data.updatedAt == request.time;
    }

    match /shared_configs/{id} {
      function okShare() {
        let d = request.resource.data;
        return d.keys().hasOnly(['owner','cfg','pro','updatedAt'])
          && d.keys().hasAll(['owner','cfg','pro','updatedAt'])
          && d.owner == request.auth.uid
          && d.cfg is map
          && d.cfg.keys().hasOnly(['name','mode','rows','cols','hearts','ids','custom','nums','range'])
          && d.pro is bool
          && d.pro == (('custom' in d.cfg) || ('nums' in d.cfg))
          && (!d.pro || subActive(request.auth.uid))
          && d.updatedAt == request.time;
      }
      allow get: if signedIn() && (!resource.data.pro || subActive(resource.data.owner));
      allow list: if false;
      allow create: if realUser() && id.matches('^[a-z0-9]{8,32}$') && okShare();
      allow update: if realUser() && resource.data.owner == request.auth.uid && okShare();
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }

    // ===== 跑吧！漢字：句庫 =====
    function okSlotKeys(s) {
      return s is map && s.keys().hasOnly(['0','1','2','3','4']);
    }

    match /teacher_banks/{uid} {
      allow read, delete: if signedIn() && request.auth.uid == uid;
      allow create: if realUser() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['slots','updatedAt'])
        && request.resource.data.keys().hasAll(['slots','updatedAt'])
        && okSlotKeys(request.resource.data.slots)
        && request.resource.data.updatedAt == request.time
        && (subActive(uid) || !request.resource.data.slots.keys().hasAny(['1','2','3','4']));
      allow update: if realUser() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['slots','updatedAt'])
        && request.resource.data.keys().hasAll(['slots','updatedAt'])
        && okSlotKeys(request.resource.data.slots)
        && request.resource.data.updatedAt == request.time
        && (subActive(uid) || request.resource.data.slots.keys().hasOnly(resource.data.slots.keys()));
    }

    match /shared_banks/{id} {
      function okBank() {
        let d = request.resource.data;
        return d.keys().hasOnly(['owner','slot','name','speed','script','boss','phonetic','sentences','pro','updatedAt'])
          && d.keys().hasAll(['owner','slot','name','speed','script','boss','sentences','pro','updatedAt'])
          && d.owner == request.auth.uid
          && d.slot is int && d.slot >= 0 && d.slot <= 4
          && d.name is string && d.name.size() >= 1 && d.name.size() <= 40
          && d.speed in ['facil','normal','dificil']
          && d.script in ['trad','simp']
          && d.boss is bool
          && (!('phonetic' in d) || d.phonetic in ['pinyin','zhuyin','both'])
          && d.sentences is list && d.sentences.size() >= 1 && d.sentences.size() <= 80
          && d.pro is bool
          && d.pro == (d.slot > 0 || d.sentences.size() > 50)
          && (!d.pro || subActive(request.auth.uid))
          && d.updatedAt == request.time;
      }

      allow get: if signedIn() && (resource.data.owner == request.auth.uid
                                   || !resource.data.pro
                                   || subActive(resource.data.owner));
      allow list: if false;

      allow create: if realUser() && id.matches('^[a-z0-9]{8,32}$') && okBank()
        && getAfter(/databases/$(database)/documents/teacher_banks/$(request.auth.uid))
             .data.slots[string(request.resource.data.slot)] == id;
      allow update: if realUser() && resource.data.owner == request.auth.uid
        && request.resource.data.slot == resource.data.slot && okBank();
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }

    // ===== 班級與成績冊（跑吧、色數、比比看）=====
    function okClassSlots(s) {
      return s is map && s.keys().hasOnly(['0','1','2']);
    }

    match /teacher_classes/{uid} {
      allow read, delete: if signedIn() && request.auth.uid == uid;
      allow create: if realUser() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['slots','updatedAt'])
        && request.resource.data.keys().hasAll(['slots','updatedAt'])
        && okClassSlots(request.resource.data.slots)
        && request.resource.data.updatedAt == request.time
        && subActive(uid);
      allow update: if realUser() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['slots','updatedAt'])
        && request.resource.data.keys().hasAll(['slots','updatedAt'])
        && okClassSlots(request.resource.data.slots)
        && request.resource.data.updatedAt == request.time
        && (subActive(uid) || request.resource.data.slots.keys().hasOnly(resource.data.slots.keys()));
    }

    match /classes/{cid} {
      function okClass() {
        let d = request.resource.data;
        return d.keys().hasOnly(['owner','slot','name','showBoard','updatedAt'])
          && d.keys().hasAll(['owner','slot','name','showBoard','updatedAt'])
          && d.owner == request.auth.uid
          && d.slot is int && d.slot >= 0 && d.slot <= 2
          && d.name is string && d.name.size() >= 1 && d.name.size() <= 40
          && d.showBoard is bool
          && d.updatedAt == request.time;
      }
      allow get: if signedIn();
      allow list: if false;
      allow create: if realUser() && cid.matches('^[a-z0-9]{8,32}$') && okClass() && subActive(request.auth.uid)
        && getAfter(/databases/$(database)/documents/teacher_classes/$(request.auth.uid))
             .data.slots[string(request.resource.data.slot)] == cid;
      allow update: if realUser() && resource.data.owner == request.auth.uid
        && request.resource.data.slot == resource.data.slot && okClass();
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }

    match /class_scores/{sid} {
      function classDoc(cid) {
        return get(/databases/$(database)/documents/classes/$(cid)).data;
      }
      function okScore() {
        let d = request.resource.data;
        return d.keys().hasOnly(['cid','owner','game','k','h','nick','best','plays','lastAt','extra'])
          && d.keys().hasAll(['cid','owner','game','k','h','nick','best','plays','lastAt'])
          && d.cid is string && d.game in ['corre','colores','dragon','ba','bei']
          && d.k is string && d.k.matches('^[a-z0-9]{1,32}$')
          && d.h is string && d.h.matches('^[a-f0-9]{16}$')
          && sid == d.cid + '_' + d.game + '_' + d.k + '_' + d.h
          && d.owner == classDoc(d.cid).owner
          && d.nick is string && d.nick.size() >= 1 && d.nick.size() <= 16
          && d.best is int && d.best >= 0 && d.best <= 1000000
          && d.plays is int
          && d.lastAt == request.time
          && (!('extra' in d) || (d.extra is map && d.extra.size() <= 6))
          && subActive(d.owner);
      }
      allow get: if signedIn() && (resource == null
                                   || resource.data.owner == request.auth.uid
                                   || classDoc(resource.data.cid).showBoard == true);
      allow list: if signedIn() && (resource.data.owner == request.auth.uid
                                    || classDoc(resource.data.cid).showBoard == true);
      allow create: if signedIn() && okScore() && request.resource.data.plays == 1;
      allow update: if signedIn() && okScore()
        && request.resource.data.cid == resource.data.cid
        && request.resource.data.game == resource.data.game
        && request.resource.data.k == resource.data.k
        && request.resource.data.h == resource.data.h
        && request.resource.data.owner == resource.data.owner
        && request.resource.data.plays == resource.data.plays + 1
        && request.resource.data.best >= resource.data.best;
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }
    match /class_plays/{pid} {
      function okPlay() {
        let d = request.resource.data;
        return d.keys().hasOnly(['cid','owner','game','h','nick','at','th','lv','rg','r','score','max','qs'])
          && d.keys().hasAll(['cid','owner','game','h','nick','at','th','lv','rg','r','score','max','qs'])
          && d.game == 'bbk'
          && d.cid is string && d.cid.matches('^[a-z0-9]{8,32}$')
          && d.owner == get(/databases/$(database)/documents/classes/$(d.cid)).data.owner
          && d.h is string && d.h.matches('^[a-f0-9]{16}$')
          && d.nick is string && d.nick.size() >= 1 && d.nick.size() <= 16
          && d.at == request.time
          && d.th is string && d.th.size() <= 20
          && d.lv is int && d.lv >= 1 && d.lv <= 3
          && d.rg is string && d.rg.size() <= 200
          && d.r is string && d.r.matches('^([a-z0-9]{8,32})?$')
          && d.max is int && d.max >= 10 && d.max <= 300
          && d.score is int && d.score >= 0 && d.score <= d.max
          && d.qs is list && d.qs.size() >= 1 && d.qs.size() <= 30
          && subActive(d.owner);
      }
      allow create: if signedIn() && okPlay();
      allow get, list: if signedIn() && resource.data.owner == request.auth.uid;
      allow update: if false;
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }

    // ===== 比比看：練習範圍 =====
    match /bbk_sets/{uid} {
      function okList() {
        let d = request.resource.data;
        let n = d.items.size();
        return d.keys().hasOnly(['items','updatedAt'])
          && d.items is list
          && d.updatedAt == request.time
          && (n <= 3
              || (n <= 15 && subActive(uid))
              || (resource != null && n <= resource.data.items.size()));
      }
      allow get: if realUser() && (request.auth.uid == uid || isAdmin());
      allow create, update: if realUser() && request.auth.uid == uid && okList();
      allow delete: if realUser() && (request.auth.uid == uid || isAdmin());
    }

    match /shared_bbk/{sid} {
      function okSet() {
        let d = request.resource.data;
        return d.keys().hasOnly(['owner','name','th','lv','d','t','n','updatedAt'])
          && d.keys().hasAll(['owner','name','th','lv','d','t','n','updatedAt'])
          && d.owner == request.auth.uid
          && d.name is string && d.name.size() >= 1 && d.name.size() <= 40
          && d.th is string && d.th.size() <= 20
          && d.lv is int && d.lv >= 1 && d.lv <= 3
          && d.d is list && d.d.size() <= 20
          && d.t is list && d.t.size() <= 20
          && d.n is int && d.n >= 3 && d.n <= 30
          && d.updatedAt == request.time;
      }
      allow get: if true;
      allow list: if false;
      allow create: if realUser() && sid.matches('^[a-z0-9]{12}$') && okSet();
      allow update: if realUser() && resource.data.owner == request.auth.uid && okSet();
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }

    // ===== 墨龍飛天：自訂主題與練習範圍 =====
    match /dragon_sets/{uid} {
      function okDragon() {
        let d = request.resource.data;
        return d.keys().hasOnly(['themes','sets','updatedAt'])
          && d.keys().hasAll(['themes','sets','updatedAt'])
          && d.themes is list && d.themes.size() <= 3
          && d.sets is list && d.sets.size() <= 15
          && d.updatedAt == request.time;
      }
      allow get: if realUser() && (request.auth.uid == uid || isAdmin());
      // 訂閱中才能新增或修改；沒訂閱（過期）只能縮減，不能變多
      allow create: if realUser() && request.auth.uid == uid && okDragon() && subActive(uid);
      allow update: if realUser() && request.auth.uid == uid && okDragon()
        && (subActive(uid)
            || (request.resource.data.themes.size() <= resource.data.themes.size()
                && request.resource.data.sets.size() <= resource.data.sets.size()));
      allow delete: if realUser() && (request.auth.uid == uid || isAdmin());
    }

    // 練習範圍連結（?d=碼）：知道碼的人讀得到單筆，不能列出全部
    match /shared_dragon/{sid} {
      function okDragonShare() {
        let d = request.resource.data;
        return d.keys().hasOnly(['owner','name','builtin','custom','updatedAt'])
          && d.keys().hasAll(['owner','name','builtin','custom','updatedAt'])
          && d.owner == request.auth.uid
          && d.name is string && d.name.size() >= 1 && d.name.size() <= 40
          && d.builtin is list && d.builtin.size() <= 10
          && d.custom is list && d.custom.size() <= 3
          && d.updatedAt == request.time;
      }
      allow get: if true;
      allow list: if false;
      allow create: if realUser() && sid.matches('^[a-z0-9]{12}$') && okDragonShare() && subActive(request.auth.uid);
      allow update: if realUser() && resource.data.owner == request.auth.uid && okDragonShare()
        && (subActive(request.auth.uid) || request.resource.data.custom.size() <= resource.data.custom.size());
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }

    // ===== 老師管理中心：意見回饋 =====
    match /center_feedback/{id} {
      function okFeedback() {
        let d = request.resource.data;
        return d.keys().hasOnly(['uid','email','type','message','lang','page','done','createdAt'])
          && d.keys().hasAll(['uid','email','type','message','lang','page','done','createdAt'])
          && d.uid == request.auth.uid
          && d.email == request.auth.token.email
          && d.type in ['idea','bug','other']
          && d.message is string && d.message.size() >= 1 && d.message.size() <= 1000
          && d.lang in ['tc','sc','es','en']
          && d.page is string && d.page.size() <= 40
          && d.done == false
          && d.createdAt == request.time;
      }
      allow create: if realUser() && okFeedback();
      allow read, delete: if isAdmin();
      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['done'])
        && request.resource.data.done is bool;
    }

    // ===== 比比看：題目回報（🚩 這題怪怪的）=====
    match /bbk_reports/{id} {
      function okReport() {
        let d = request.resource.data;
        return d.keys().hasOnly(['why','note','type','d','f','ids','q','ans','pick','lv','r','th','lang','ver','done','at'])
          && d.keys().hasAll(['why','note','type','d','f','ids','q','ans','pick','lv','r','th','lang','ver','done','at'])
          && d.why in ['wrong','image','other']
          && d.note is string && d.note.size() <= 100
          && d.type is string && d.type.size() <= 10
          && d.d is string && d.d.size() <= 20
          && d.f is string && d.f.size() <= 10
          && d.ids is list && d.ids.size() >= 1 && d.ids.size() <= 6
          && d.q is string && d.q.size() <= 300
          && d.ans is string && d.ans.size() <= 100
          && d.pick is string && d.pick.size() <= 100
          && d.lv is int && d.lv >= 1 && d.lv <= 3
          && d.r is string && d.r.matches('^([a-z0-9]{8,32})?$')
          && d.th is string && d.th.size() <= 20
          && d.lang in ['es','en','zh']
          && d.ver is string && d.ver.size() <= 12
          && d.done == false
          && d.at == request.time;
      }
      allow create: if signedIn() && okReport();
      allow read, delete: if isAdmin();
      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['done'])
        && request.resource.data.done is bool;
    }
  }
}
